scn ALTBackgroundMenu

int iButton
string_var sMenu
int iIndex
string_var sMSG
string_var sMSGDesc
string_var sMSGTitle
int iBut
ref rMSG
int iGo
ref rBackMsg

array_var aChoices
int iCount
int iNext
int iPrevious

begin function {iButton}

    ; The old hover panel treats button positions as background IDs.
    ; Disable it because these menus use an explicit ID mapping.
    set ALTStartQ.bCustomMenu to -1

    if iButton == -1
        set ALTStartQ.iUI to -1
        let rMSG := (begin function {iBut}
            if iBut == 0
                call ALTBackgroundMenu 100
            elseif iBut == 1
                call ALTBackgroundMenu 200
            elseif iBut == 2
                call ALTBackgroundMenu 300
            elseif iBut == 3
                call ALTBackgroundMenu 0
            endif
        end)
        MessageBoxExAlt rMSG "^NVO - Choose Your Start^Choose a category. Each background keeps its original equipment, location and consequences. Becoming the Courier remains optional: sign up at a Mojave Express flier later.|Roleplay|Hardcore|Powerful|Random - any background"
        return
    endif

    ; Original background IDs remain unchanged: 0 through 58.
    if iButton >= 0 && iButton <= 58
        let rBackMsg := ListGetNthForm ALTBackList iButton
        if IsFormValid rBackMsg == 0
            call ALTBackgroundMenu -1
            return
        endif
        let sMSGDesc := GetFormDescription rBackMsg
        let sMSGTitle := LNGetName rBackMsg
        set ALTStartQ.iBackground to iButton
        let rMSG := (begin function {iBut}
            if iBut == 0
                set ALTBackground to ALTStartQ.iBackground
                set ALTStartQ.iStage to 1
            else
                call ALTBackgroundMenu ALTStartQ.iUI
            endif
        end)
        let sMSG := "^" + sMSGTitle + "^" + sMSGDesc
        let sMSG += "|Choose This Background|Back"
        MessageBoxExAlt rMSG "%z" sMSG
        return
    endif

    let iNext := -1
    let iPrevious := -1

    if iButton == 100
        let aChoices := Ar_List 1, 2, 19, 20, 35, 18, 40
        let sMenu := "^Roleplay - 1/5^Travelers, soldiers and familiar wasteland origins. Select a background to read its description."
        let iNext := 101
    elseif iButton == 101
        let aChoices := Ar_List 7, 27, 12, 16, 24, 13, 8
        let sMenu := "^Roleplay - 2/5^Faction ties and local identities. Read the description for benefits and enemies."
        let iPrevious := 100
        let iNext := 102
    elseif iButton == 102
        let aChoices := Ar_List 4, 5, 10, 14, 25, 36, 37
        let sMenu := "^Roleplay - 3/5^Jobs, settlements and independent lives in the Mojave."
        let iPrevious := 101
        let iNext := 103
    elseif iButton == 103
        let aChoices := Ar_List 38, 39, 41, 44, 45, 46, 47
        let sMenu := "^Roleplay - 4/5^Unusual histories and specialist interests. Some backgrounds have significant drawbacks."
        let iPrevious := 102
        let iNext := 104
    elseif iButton == 104
        let aChoices := Ar_List 48, 49, 50, 51, 52
        let sMenu := "^Roleplay - 5/5^Technical careers, bounty work and established communities."
        let iPrevious := 103
    elseif iButton == 200
        let aChoices := Ar_List 3, 9, 11, 15, 29, 31, 32
        let sMenu := "^Hardcore - 1/3^Hardship and dangerous openings. This category does not automatically enable Hardcore survival mode."
        let iNext := 201
    elseif iButton == 201
        let aChoices := Ar_List 21, 22, 23, 30, 33, 42, 54
        let sMenu := "^Hardcore - 2/3^Outlaws, fugitives and hostile surroundings. Check each background's consequences before choosing."
        let iPrevious := 200
        let iNext := 202
    elseif iButton == 202
        let aChoices := Ar_List 55, 56, 57, 58
        let sMenu := "^Hardcore - 3/3^DLC openings. These use the original mod's special DLC starts and may place you in dangerous or restricted areas."
        let iPrevious := 201
    elseif iButton == 300
        let aChoices := Ar_List 6, 17, 26, 28, 34, 43, 53
        let sMenu := "^Powerful^Start with useful equipment, perks, allies or wealth. Original drawbacks still apply."
    else
        call ALTBackgroundMenu -1
        return
    endif

    ; Remember the category page for the preview's Back button.
    set ALTStartQ.iUI to iButton
    let iCount := Ar_Size aChoices
    let iIndex := 0
    while iIndex < iCount
        let iGo := aChoices[iIndex]
        let rBackMsg := ListGetNthForm ALTBackList iGo
        if IsFormValid rBackMsg == 0
            break
        endif
        let sMSGTitle := LNGetName rBackMsg
        let sMenu += "|" + sMSGTitle
        let iIndex += 1
    loop

    if iIndex < iCount
        call ALTBackgroundMenu -1
        return
    endif

    ; Navigation buttons use the same mapping as background buttons.
    if iPrevious != -1
        Ar_Append aChoices iPrevious
        let sMenu += "|Previous page"
    endif
    if iNext != -1
        Ar_Append aChoices iNext
        let sMenu += "|Next page"
    endif
    Ar_Append aChoices, -1
    let sMenu += "|Back to categories"

    let rMSG := (begin function {iBut}
        if eval iBut >= 0 && iBut < (Ar_Size aChoices)
            let iGo := aChoices[iBut]
            call ALTBackgroundMenu iGo
        endif
    end)
    MessageBoxExAlt rMSG "%z" sMenu
end
