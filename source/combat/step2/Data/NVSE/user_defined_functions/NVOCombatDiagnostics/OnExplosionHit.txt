ref rTarget
ref rSource
ref rExplosion
ref rExplosionBase
ref rTargetBase
ref rWriter
float fDamage
string_var sLine

; ShowOff calls ON the explosion. Parameters are target actor, then source.
begin function {rTarget, rSource}
    if eval (player.AuxVarGetFlt "*_NVODiagEnabled") == 0
        return
    endif
    set rExplosion to GetSelfAlt
    if IsFormValid rExplosion == 0
        return
    endif
    set fDamage to GetExplosionHitDamage
    set rExplosionBase to rExplosion.GetBaseObject
    set rTargetBase to 0
    if IsFormValid rTarget
        set rTargetBase to rTarget.GetBaseObject
    endif

    let sLine := sv_construct "EXPLOSION explosion=%i explosion_base=%i target=%i target_base=%i source=%i explosion_damage=%.3f" rExplosion rExplosionBase rTarget rTargetBase rSource fDamage
    set rWriter to player.AuxVarGetRef "*_NVODiagWriter"
    call rWriter sLine
    sv_destruct sLine
end
